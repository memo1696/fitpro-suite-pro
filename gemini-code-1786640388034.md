      res.innerHTML = `<div style="background:#18181b; padding:15px; border-radius:8px; border:1px solid #27272a;"><p style="margin:0; color:#fff;">🔥 Calorías objetivo: <strong>${Math.round(cal)} kcal</strong></p><p style="margin:5px 0 0 0; color:#a1a1aa; font-size:13px;">Proteína recomendada: ${Math.round(p * 2.2)}g diarios</p></div>`;
    }

    function renderizarBiblioteca(cat) {
      const grid = document.getElementById('library-grid');
      if(!grid) return;
      grid.innerHTML = '';
      const filtrados = cat === 'todos' ? ejerciciosDB : ejerciciosDB.filter(e => e.categoria === cat);
      filtrados.forEach(ej => {
        grid.innerHTML += `<div style="background:#121214; border:1px solid #26262b; border-radius:12px; padding:20px; display:flex; flex-direction:column;"><span style="color:#22c55e; font-size:12px; font-weight:bold; text-transform:uppercase; margin-bottom:8px;">${ej.categoria}</span><h3 style="color:#fff; margin:0 0 8px 0; font-size:18px;">${ej.nombre}</h3><div style="background:#18181b; padding:12px; border-radius:8px; margin-top:auto;"><p style="color:#d4d4d8; font-size:13px; margin:0 0 5px 0;"><strong>🎯 Músculos:</strong> ${ej.musculos}</p><p style="color:#a1a1aa; font-size:12px; margin:0;"><strong>📝 Técnica:</strong> ${ej.ejecucion}</p></div></div>`;
      });
    }
